211 static Bytes fromHexString(const String &hexStr) {
212 size_t len = hexStr.length();
213 if (len % 2 != 0) {
214 UNIOT_LOG_ERROR("invalid hex string length");
215 return Bytes();
216 }
217
218 Bytes bytes;
219 if (!bytes._reserve(len / 2)) {
220 return Bytes();
221 }
222 for (size_t i = 0; i < len; i += 2) {
223 char buf[3] = {hexStr.charAt(i), hexStr.charAt(i + 1), '\0'};
224 uint8_t b = strtol(buf, nullptr, 16);
225 bytes.mBuffer[i / 2] = b;
226 }
227 return bytes;
228 }

400 bool _reserve(size_t newSize) {
401 // realloc(ptr, 0) may free and return null, which would read as a failure below.
402 if (!newSize) {
403 _invalidate();
404 return true;
405 }
406
407 // On failure realloc leaves the original block intact, so the object is left exactly
408 // as it was: never a size the buffer does not have, and nothing leaked.
409 auto buffer = (uint8_t *)realloc(mBuffer, newSize);
410 if (!buffer) {
411 return false;
412 }
413
414 if (newSize > mSize) {
415 memset(buffer + mSize, 0, newSize - mSize);
416 }
417 mBuffer = buffer;
418 mSize = newSize;
419 return true;
420 }

431 Bytes &_copy(const uint8_t *data, size_t size) {
432 if (_reserve(size)) {
433 if (size) {
434 memcpy(mBuffer, data, size);
435 }
436 } else {
437 _invalidate();
438 }
439 return *this;
440 }
