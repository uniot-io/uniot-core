489 int addLispButton(uint8_t pin, uint8_t activeLevel = LOW, uint32_t id = FOURCC(_btn), uniot::Button::ButtonCallback callback = nullptr) {
490 constexpr uint8_t longPressTicks = 30;
491 constexpr uint32_t pollMs = 100;
492
493 auto button = uniot::MakeShared<uniot::Button>(pin, activeLevel, longPressTicks, callback);
494
495 auto &registers = uniot::PrimitiveExpeditor::getRegisterManager();
496 if (!registers.link(uniot::primitive::name::bclicked, button.get(), id)) {
497 return -1;
498 }
499 mButtons.push(button);
500
501 // One task polls every button, created with the first.
502 if (!mpTaskButtons) {
503 mpTaskButtons = uniot::TaskScheduler::make([this](uniot::SchedulerTask &, short times) {
504 mButtons.forEach([times](const uniot::SharedPointer<uniot::Button> &button) {
505 button->execute(times);
506 });
507 });
508 mScheduler.push("lisp_buttons", mpTaskButtons);
509 mpTaskButtons->attach(pollMs);
510 }
511
512 return static_cast<int>(registers.getRegisterLength(uniot::primitive::name::bclicked)) - 1;
513 }

562 TimerId setInterval(std::function<void()> callback, uint32_t intervalMs, short times = 0) {
563 if (!callback) {
564 return INVALID_TIMER_ID;
565 }
566
567 auto id = _generateTimerId();
568 auto task = uniot::TaskScheduler::make([this, id, callback = std::move(callback)](uniot::SchedulerTask&, short remainingTimes) {
569 callback();
570
571 if (!remainingTimes) {
572 mActiveTimers.remove(id);
573 }
574 });
575
576 mActiveTimers.put(id, task);
577 mScheduler.push(nullptr, task);
578 task->attach(intervalMs, times);
579 return id;
580 }

681 template <typename... Topics>
682 ListenerId addSystemListener(std::function<void(unsigned int, int)> callback, unsigned int firstTopic, Topics... otherTopics) {
683 if (!callback) {
684 return INVALID_LISTENER_ID;
685 }
686
687 unsigned int topics[] = {firstTopic, static_cast<unsigned int>(otherTopics)...};
688 size_t count = sizeof...(otherTopics) + 1;
689
690 auto listener = uniot::MakeShared<uniot::CoreCallbackEventListener>(callback);
691 for (size_t i = 0; i < count; ++i) {
692 listener->listenToEvent(topics[i]);
693 }
694
695 if (mEventBus.registerEntity(listener.get())) {
696 auto id = _generateListenerId();
697 mActiveListeners.put(id, listener);
698 return id;
699 }
700
701 return INVALID_LISTENER_ID;
702 }
703
711 bool removeSystemListener(ListenerId id) {
712 if (id == INVALID_LISTENER_ID) {
713 return false;
714 }
715
716 auto listener = mActiveListeners.get(id, nullptr);
717 if (listener) {
718 mEventBus.unregisterEntity(listener.get());
719 return mActiveListeners.remove(id);
720 }
721 return false;
722 }

733 template <typename... Topics>
734 size_t removeSystemListeners(unsigned int firstTopic, Topics... otherTopics) {
735 unsigned int topics[] = {firstTopic, static_cast<unsigned int>(otherTopics)...};
736 size_t topicCount = sizeof...(otherTopics) + 1;
737 size_t removedCount = 0;
738
739 mActiveListeners.begin();
740 while (!mActiveListeners.isEnd()) {
741 auto listener = mActiveListeners.current().second;
742 bool shouldRemove = false;
743
744 if (listener) {
745 for (size_t i = 0; i < topicCount; ++i) {
746 if (listener->isListeningToEvent(topics[i])) {
747 shouldRemove = true;
748 break;
749 }
750 }
751 }
752
753 if (shouldRemove) {
754 mEventBus.unregisterEntity(listener.get());
755 mActiveListeners.deleteCurrent();
756 removedCount++;
757 } else {
758 mActiveListeners.next();
759 }
760 }
761
762 return removedCount;
763 }

839 void begin(uint32_t eventBusTaskPeriod = 10) {
840 UNIOT_LOG_SET_READY();
841
842 auto& app = uniot::AppKit::getInstance();
843
844 if (mpNetworkControllerConfig) {
845 app.configureNetworkController(*mpNetworkControllerConfig);
846 _deleteNetworkControllerConfig();
847 }
848
849 mEventBus.registerKit(app);
850
851 auto taskHandleEventBus = uniot::TaskScheduler::make(mEventBus);
852 mScheduler.push("event_bus", taskHandleEventBus);
853 taskHandleEventBus->attach(eventBusTaskPeriod);
854
855 // Registering a pin for dread or aread sets it to plain INPUT, which clears any pull. Every
856 // button gets its own mode back here, so the order of those calls in setup() does not
857 // matter. Before attach(), which runs the stored script.
858 _forEachLispButton([](uniot::Button *button) {
859 button->applyPinMode();
860 });
861
862 mScheduler.push(app);
863 app.attach();
864 }
