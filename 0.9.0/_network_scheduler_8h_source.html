327 void _initTasks() {
328 // Server lifecycle tasks
329 mTaskStart = TaskScheduler::make([this](SchedulerTask &self, short t) {
330 mTaskStop->detach();
331 if (mConfigServer.start()) {
332 _initServerCallbacks();
333 mConfigServer.wsEnable(true); // Ensure that WS are enabled after disabling them in "Step 1 of Stopping Configuration" during the AP recovery process
334 mTaskServe->attach(10);
335 } else {
336 UNIOT_LOG_WARN("config server failed to start, restarting");
337 self.once(1000);
338 }
339 });
340
341 mTaskServe = TaskScheduler::make(mConfigServer);
342
343 mTaskStop = TaskScheduler::make([this](SchedulerTask &self, short t) {
344 static bool wsClosed = false;
345 UNIOT_LOG_DEBUG("stopping config server, state: %d", wsClosed);
346 // 1: close websocket
347 // 2: stop access point
348 // 3: stop server
349 if (!wsClosed) {
350 mConfigServer.wsCloseAll();
351 wsClosed = true;
352 self.once(10000); // Stopping Configuration. Step 3. Carefully change the deferrals.
353 return;
354 }
355 mTaskServe->detach();
356 mConfigServer.stop();
357 wsClosed = false;
358 mLastNetworks = static_cast<const char *>(nullptr); // invalidate String
359 });
360
361 // Access Point tasks
362 mTaskConfigAp = TaskScheduler::make([this](SchedulerTask &self, short t) {
363 WiFi.disconnect(true, true);
364 mTaskStopAp->detach();
365 if (WiFi.softAPConfig(mConfigServer.ip(), mConfigServer.ip(), mApSubnet) && WiFi.softAP(mApName.c_str(), mApPassword.isEmpty() ? nullptr : mApPassword.c_str())) {
366#if defined(ESP32) && defined(ENABLE_LOWER_WIFI_TX_POWER)
367 WiFi.setTxPower(WIFI_TX_POWER_LEVEL);
368#endif
369 mTaskStart->once(500);
370 mTaskScan->once(500);
371 mTaskAvailabilityCheck->attach(10000);
372 mApEnabled = true;
373 CoreEventEmitter::sendDataToChannel(events::network::Channel::OUT_SSID, Bytes(mApName));
374 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::ACCESS_POINT);
375 } else {
376 UNIOT_LOG_WARN("config server failed to start");
377 mTaskConfigAp->attach(500, 1);
378 }
379 });
380
381 mTaskStopAp = TaskScheduler::make([this](SchedulerTask &self, short t) {
382 mApEnabled = false;
383 WiFi.softAPdisconnect(true); // check with 8266
384 });
385
386 // Station connection tasks
387 mTaskConnectSta = TaskScheduler::make([this](SchedulerTask &self, short t) {
388 WiFi.disconnect(false, true);
389#if defined(ESP32)
390 if (mUseFallbackScan) {
391 UNIOT_LOG_INFO("connecting with WIFI_FAST_SCAN (fallback)");
392 WiFi.setScanMethod(WIFI_FAST_SCAN);
393 } else {
394 WiFi.setScanMethod(WIFI_ALL_CHANNEL_SCAN);
395 }
396#endif
397 bool connect = WiFi.begin(mWifiStorage.getSsid().c_str(), mWifiStorage.getPassword().c_str()) != WL_CONNECT_FAILED;
398 if (connect) {
399#if defined(ESP32) && defined(ENABLE_LOWER_WIFI_TX_POWER)
400 WiFi.setTxPower(WIFI_TX_POWER_LEVEL);
401#endif
402 mTaskConnecting->attach(100, 100);
403 CoreEventEmitter::sendDataToChannel(events::network::Channel::OUT_SSID, Bytes(mWifiStorage.getSsid()));
404 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::CONNECTING);
405 mCanScan = false;
406 mLastSaveResult = -1;
407 } else {
408 mTaskConnecting->detach();
409 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::FAILED);
410 mCanScan = true;
411 mLastSaveResult = 0;
412 }
413 });
414
415 mTaskConnecting = TaskScheduler::make([this](SchedulerTask &self, short times) {
416 auto __processFailure = [this](int triesBeforeGivingUp = 3) {
417 static int tries = 0;
418 if (++tries < triesBeforeGivingUp) {
419 UNIOT_LOG_INFO("attempts left before giving up: %d", triesBeforeGivingUp - tries);
420 mTaskConnectSta->attach(500, 1);
421 } else {
422 tries = 0;
423#if defined(ESP32)
424 if (!mUseFallbackScan) {
425 UNIOT_LOG_WARN("all-channel scan did not associate, retrying with fast scan");
426 mUseFallbackScan = true;
427 mTaskConnectSta->attach(500, 1);
428 return;
429 }
430 mUseFallbackScan = false;
431#endif
432 mWifiStorage.restore();
433 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::FAILED);
434 mCanScan = true;
435 mLastSaveResult = 0;
436 }
437 };
438
439 switch (WiFi.status()) {
440 case WL_CONNECTED:
441 self.detach();
442 mTaskMonitoring->attach(200);
443 mWifiStorage.store();
444 if (mpCredentials->isOwnerChanged()) {
445 mpCredentials->store();
446 }
447
448 mCanScan = true;
449 mLastSaveResult = 1;
450 mTaskStop->once(30000); // Stopping Configuration. Step 1. Carefully change the deferrals.
451 mTaskStopAp->once(35000); // Stopping Configuration. Step 2. Carefully change the deferrals.
452 mTaskAvailabilityCheck->detach();
453#if defined(ESP32)
454 mUseFallbackScan = false;
455#if UNIOT_WIFI_NO_SLEEP
456 // Keep the radio awake between beacons; see UNIOT_WIFI_NO_SLEEP in Common.h.
457 WiFi.setSleep(false);
458#endif // UNIOT_WIFI_NO_SLEEP
459#endif // ESP32
460 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::SUCCESS);
461 break;
462
463 case WL_NO_SSID_AVAIL:
464 case WL_CONNECT_FAILED:
465 self.detach();
466 __processFailure();
467 break;
468#if defined(ESP8266)
469 case WL_WRONG_PASSWORD:
470 self.detach();
471 __processFailure(1);
472 break;
473#endif
474 case WL_IDLE_STATUS:
475 case WL_DISCONNECTED:
476 case WL_CONNECTION_LOST:
477 if (!times) {
478 __processFailure();
479 }
480 break;
481
482 default:
483 UNIOT_LOG_WARN("unexpected WiFi status: %d", WiFi.status());
484 break;
485 }
486 });

488 mTaskMonitoring = TaskScheduler::make([this](SchedulerTask &self, short times) {
489 if (WiFi.status() != WL_CONNECTED) {
490 mTaskMonitoring->detach();
491 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::DISCONNECTED);
492 }

495 // Network scanning tasks
496 mTaskScan = TaskScheduler::make([this](SchedulerTask &self, short times) {
497 static auto __broadcastNets = [this](const String &netJsonArray) {
498 String nets;
499 JSON::Object(nets)
500 .put("nets", netJsonArray, false)
501 .close();
502 mConfigServer.wsTextAll(nets);
503 delay(50); // to allow for all clients to receive the message (relevant for ESP32)
504 };
505 if (mCanScan) {
506 mWifiScan.scanNetworksAsync([this](int n) {
507 mLastNetworks = static_cast<const char *>(nullptr); // invalidate String
508 JSON::Array jsonNets(mLastNetworks);
509 for (auto i = 0; i < n; ++i) {
510 jsonNets.appendArray()
511 .append(WiFi.BSSIDstr(i))
512 .append(WiFi.SSID(i))
513 .append(WiFi.RSSI(i))
514 .append(mWifiScan.isSecured(WiFi.encryptionType(i)))
515 .close();
516 }
517 jsonNets.close();
518 WiFi.scanDelete();
519 __broadcastNets(mLastNetworks);
520 });
521 } else {
522 __broadcastNets(mLastNetworks);
523 }
524 });

526 mTaskAvailabilityCheck = TaskScheduler::make([this](SchedulerTask &self, short times) {
527 static int scanInProgressFuse = 0;
528 if (scanInProgressFuse-- > 0) {
529 UNIOT_LOG_INFO("availability check skipped, scan in progress");
530 return;
531 }
532
533 if (mCanScan &&
534 !mConfigServer.wsClientsActive() &&
535 mWifiStorage.isCredentialsValid()) {
536 UNIOT_LOG_INFO("checking availability of network '%s'", mWifiStorage.getSsid().c_str());
537 scanInProgressFuse = 3;
538
539 mWifiScan.scanNetworksAsync([&](int n) {
540 scanInProgressFuse = 0;
541 if (self.isAttached() &&
542 mCanScan &&
543 !mConfigServer.wsClientsActive() &&
544 mWifiStorage.isCredentialsValid()) {
545 for (auto i = 0; i < n; ++i) {
546 if (WiFi.SSID(i) == mWifiStorage.getSsid()) {
547 UNIOT_LOG_INFO("network '%s' is available", WiFi.SSID(i).c_str());
548 CoreEventEmitter::emitEvent(events::network::Topic::CONNECTION, events::network::Msg::AVAILABLE);
549 break;
550 }
551 }
552 } else {
553 UNIOT_LOG_INFO("scan done, skipping availability check");
554 }
555 WiFi.scanDelete();
556 });
557 }
558 });

596 void _handleWebSocketEvent(AsyncWebSocket *server, AsyncWebSocketClient *client, AwsEventType type, void *arg, uint8_t *data, size_t len) {
597 switch (type) {
598 case WS_EVT_CONNECT:
599 // mAppState.Network.WebSocketsClients = mWebSocket.count();
600 UNIOT_LOG_INFO("WebSocket client #%u connected from %s", client->id(), client->remoteIP().toString().c_str());
601 break;
602 case WS_EVT_DISCONNECT:
603 // mAppState.Network.WebSocketsClients = mWebSocket.count();
604 UNIOT_LOG_INFO("WebSocket client #%u disconnected", client->id());
605 break;
606 case WS_EVT_DATA:
607 _handleWebSocketMessage(client->id(), arg, data, len);
608 break;
609 default:
610 break;
611 }

625 void _handleWebSocketMessage(uint32_t clientId, void *arg, uint8_t *data, size_t len) {
626 auto *info = (AwsFrameInfo *)arg;
627 if (info->opcode == WS_BINARY) {
628 CBORObject msg(Bytes(data, len));
629 if (!msg.hasError()) {
630 auto action = msg.getInt("action");
631 if (action != ACTIONS::INVALID) {
632 switch (action) {
633 case ACTIONS::STATUS: {
634 String status;
635 JSON::Object(status)
636 .put("id", mpCredentials->getDeviceId())
637 .put("acc", mpCredentials->getOwnerId())
638 .put("nets", mLastNetworks.length() ? mLastNetworks : "[]", false)
639 .put("homeNet", WiFi.isConnected() ? WiFi.SSID() : "")
640 .put("customApp", mCustomPagePath)
641 .put("customAppLabel", mCustomPageLabel)
642 .close();
643 mConfigServer.wsTextAll(status);
644 break;
645 }
646 case ACTIONS::SAVE: {
647 mWifiStorage.setCredentials(msg.getString("ssid"), msg.getString("pass"));
648 if (mWifiStorage.isCredentialsValid()) {
649 mTaskConnectSta->once(500);
650 mpCredentials->setOwnerId(msg.getString("acc"));
651 UNIOT_LOG_DEBUG("owner changed: %d", mpCredentials->isOwnerChanged());
652 }
653 break;
654 }
655 case ACTIONS::SCAN: {
656 mTaskScan->once(1000);
657 break;
658 }
659 case ACTIONS::ASK: {
660 if (mLastSaveResult > -1) {
661 String success;
662 JSON::Object(success)
663 .put("success", mLastSaveResult)
664 .close();
665 mConfigServer.wsText(clientId, success);
666 }
667 break;
668 }
669 case ACTIONS::PING:
670 // Nothing to answer. Receiving any message already refreshes the
671 // client's last-seen timestamp, which keeps wsClientsActive()
672 // true and suppresses background WiFi scans while a page is open.
673 break;
674 default:
675 break;
676 }
677 } else {
678 UNIOT_LOG_WARN("WebSocket message is not a valid action");
679 }
680 } else {
681 UNIOT_LOG_WARN("WebSocket message is not a valid CBOR");
682 }
683 }
