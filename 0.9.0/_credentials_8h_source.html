93 virtual bool store() override {
94 object().put("account", mOwnerId.c_str());
95 object().put("private_key", mPrivateKey.raw(), mPrivateKey.size());
96 return CBORStorage::store();
97 }
98
107 virtual bool restore() override {
108 if (CBORStorage::restore()) {
109 mOwnerId = object().getString("account");
110 mPrivateKey = object().getBytes("private_key");
111
112 // Every use of the key reads exactly PRIVATE_KEY_SIZE bytes from it, so a stored key of
113 // any other length is dropped rather than read past. The constructor then generates a
114 // new one, which gives the device a new identity: the platform sees a new public key.
115 if (mPrivateKey.size() && mPrivateKey.size() != PRIVATE_KEY_SIZE) {
116 UNIOT_LOG_ERROR("stored private key is %u bytes, expected %u, discarding it",
117 mPrivateKey.size(), PRIVATE_KEY_SIZE);
118 mPrivateKey.clean();
119 }
120 return true;
121 }
122 UNIOT_LOG_ERROR("%s", "credentials not restored");
123 return false;
124 }

215 virtual Bytes sign(const Bytes &data) const override {
216 if (mPrivateKey.size() != PRIVATE_KEY_SIZE || mPublicKeyRaw.size() != PUBLIC_KEY_SIZE) {
217 UNIOT_LOG_ERROR("%s", "cannot sign without a valid key pair");
218 return Bytes();
219 }
220
221 // The public key is the one derived at startup. Deriving it again here would repeat the
222 // scalar multiplication that is the expensive half of Ed25519, on every signature.
223 uint8_t signature[SIGNATURE_SIZE];
224 Ed25519::sign(signature, mPrivateKey.raw(), mPublicKeyRaw.raw(), data.raw(), data.size());
225 return Bytes(signature, sizeof(signature));
226 }
227
233 virtual COSEAlgorithm signerAlgorithm() const override {

245 String _calcDeviceId() {
246 uint8_t mac[6];
247 char macStr[13] = {0};
248#if defined(ESP8266)
249 wifi_get_macaddr(STATION_IF, mac);
250#elif defined(ESP32)
251 esp_read_mac(mac, ESP_MAC_WIFI_STA);
252#endif
253 for (uint8_t i = 0; i < 6; i++)
254 sprintf(macStr + i * 2, "%02x", mac[i]);
255
256 return String(macStr);
257 }
258
264 void _generatePrivateKey() {
265 // Initialize the random number generator with device-specific entropy
266 RNG.begin(String("uniot::entropy::" + mCreatorId + "::" + mDeviceId).c_str());
267
268 uint8_t privateKey[PRIVATE_KEY_SIZE];
269 Ed25519::generatePrivateKey(privateKey);
270 mPrivateKey = Bytes(privateKey, sizeof(privateKey));
271 }
272
278 void _derivePublicKey() {
279 if (mPrivateKey.size() != PRIVATE_KEY_SIZE) {
280 UNIOT_LOG_ERROR("%s", "no valid private key to derive from");
281 return;
282 }
283
284 uint8_t publicKey[PUBLIC_KEY_SIZE];
285 Ed25519::derivePublicKey(publicKey, mPrivateKey.raw());
286 mPublicKeyRaw = Bytes(publicKey, sizeof(publicKey));
287 mPublicKey = mPublicKeyRaw.toHexString();
288 }
