225inline uint32_t CRC32(const void *data, size_t length, uint32_t crc = 0) {
226 const uint8_t *ldata = (const uint8_t *)data;
227
228 crc = ~crc;
229 while (length--) {
230 crc ^= *ldata++;
231 for (uint8_t k = 0; k < 8; k++)
232 crc = crc & 1 ? (crc >> 1) ^ 0x82f63b78 : crc >> 1; // CRC-32C (iSCSI) polynomial in reversed bit order.
233 }
234 return ~crc;
235}
