159 virtual void execute(short times) override {
160 bool curState = digitalRead(mPin) == mActiveLevel;
161 // The counter stops at the threshold. It is a uint8_t, so left running it would wrap
162 // after 255 ticks: a long hold would fire LONG_PRESS again, and releasing just after the
163 // wrap would count as a click.
164 if (curState && mLongPressTicker < mLongPressTicks) {
165 if (++mLongPressTicker == mLongPressTicks) {
166 mWasLongPress = true;
167 if (OnLongPress)
168 OnLongPress(this, LONG_PRESS);
169 }
170 }
171 if (mPrevState && !curState) {
172 if (mLongPressTicker < mLongPressTicks) {
173 mWasClick = true;
174 if (OnClick)
175 OnClick(this, CLICK);
176 }
177 mLongPressTicker = 0;
178 mAutoResetTicker = 0;
179 }
180 mPrevState = curState;
181
182 if (++mAutoResetTicker == mAutoResetTicks) {
183 resetClick();
184 resetLongPress();
185 mAutoResetTicker = 0;
186 }
187 }

206 static uint8_t _inputMode(uint8_t pin, uint8_t activeLevel) {
207#if defined(ESP8266)
208 if (pin == 16) {
209 return activeLevel == HIGH ? INPUT_PULLDOWN_16 : INPUT;
210 }
211 return activeLevel == LOW ? INPUT_PULLUP : INPUT;
212#elif defined(ESP32)
213 // Input-only pins have no pull resistors, and asking for one can fail the whole config.
214 if (!GPIO_IS_VALID_OUTPUT_GPIO(pin)) {
215 return INPUT;
216 }
217 return activeLevel == LOW ? INPUT_PULLUP : INPUT_PULLDOWN;
218#else
219 return INPUT;
220#endif
221 }
