244 eventBus.openDataChannel(events::lisp::Channel::OUT_EVENT, 10);

246 eventBus.registerEntity(&Date::getInstance());

258 UNIOT_LOG_WARN("configure the network controller before registering to the event bus");

453 inline void _initPrimitives() {
454 if (PrimitiveExpeditor::getRegisterManager().getRegisterLength(primitive::name::dwrite)) {
455 getLisp().pushPrimitive(primitive::dwrite);
456 }
457 if (PrimitiveExpeditor::getRegisterManager().getRegisterLength(primitive::name::dread)) {
458 getLisp().pushPrimitive(primitive::dread);
459 }
460 // Analog values are 10 bits on every chip, so a script reads and writes the same range
461 // wherever it runs. These are global Arduino settings, so they are only applied when a
462 // script can actually use analog, and a sketch that wants different values sets them
463 // after begin().
464 if (PrimitiveExpeditor::getRegisterManager().getRegisterLength(primitive::name::awrite)) {
465 getLisp().pushPrimitive(primitive::awrite);
466#if defined(ESP8266)
467 analogWriteRange(1023);
468#elif defined(ESP32)
469 analogWriteResolution(10);
470#endif
471 }
472 if (PrimitiveExpeditor::getRegisterManager().getRegisterLength(primitive::name::aread)) {
473 getLisp().pushPrimitive(primitive::aread);
474 // The ESP8266's ADC is 10 bits already; the ESP32 family returns 12 or 13.
475#if defined(ESP32)
476 analogReadResolution(10);
477#endif
478 }
479 if (PrimitiveExpeditor::getRegisterManager().getRegisterLength(primitive::name::bclicked)) {
480 getLisp().pushPrimitive(primitive::bclicked);
481 }
482 }

490 inline void _initListeners() {
491 mpNetworkEventListener = MakeUnique<CoreCallbackEventListener>([&](int topic, int msg) {
492 if (events::network::Topic::CONNECTION == topic) {
493 switch (msg) {
494 case events::network::Msg::SUCCESS:
495 UNIOT_LOG_DEBUG("network connected, ip: %s", WiFi.localIP().toString().c_str());
496 break;
497 case events::network::Msg::ACCESS_POINT:
498 UNIOT_LOG_DEBUG("access point mode");
499 mpNetworkEventListener->receiveDataFromChannel(events::network::Channel::OUT_SSID, [this](unsigned int id, bool empty, Bytes data) {
500 if (!empty) {
501 UNIOT_LOG_DEBUG("SSID: %s", data.terminate().c_str());
502 }
503 });
504 break;
505
506 case events::network::Msg::CONNECTING:
507 UNIOT_LOG_DEBUG("connecting");
508 mpNetworkEventListener->receiveDataFromChannel(events::network::Channel::OUT_SSID, [this](unsigned int id, bool empty, Bytes data) {
509 if (!empty) {
510 UNIOT_LOG_DEBUG("SSID: %s", data.terminate().c_str());
511 }
512 });
513 break;
514
515 case events::network::Msg::DISCONNECTING:
516 UNIOT_LOG_DEBUG("disconnecting");
517 break;
518
519 case events::network::Msg::DISCONNECTED:
520 UNIOT_LOG_DEBUG("disconnected");
521 break;
522
523 case events::network::Msg::AVAILABLE:
524 UNIOT_LOG_DEBUG("network available");
525 break;
526
527 case events::network::Msg::FAILED:
528 default:
529 UNIOT_LOG_DEBUG("connection failed");
530 break;
531 }
532 return;
533 }
534 if (events::mqtt::Topic::CONNECTION == topic) {
535 switch (msg) {
536 case events::mqtt::Msg::SUCCESS:
537 UNIOT_LOG_DEBUG("MQTT connected");
538 if (mCredentials.isOwnerChanged()) {
539 UNIOT_LOG_INFO("owner changed, renewing subscriptions");
540 mMQTT.renewSubscriptions();
541 mCredentials.resetOwnerChanged();
542 } else {
543 UNIOT_LOG_INFO("owner not changed, keeping subscriptions");
544 }
545 break;
546 case events::mqtt::Msg::FAILED:
547 default:
548 UNIOT_LOG_DEBUG("MQTT connection failed");
549 break;
550 }
551 return;
552 }
553 });
554 }
