167/*
168 * The file and the line are literals, so they are concatenated into the format and go to
169 * flash with it. __func__ is a variable, not a literal, so it stays an argument in RAM.
170 */
171#define UNIOT_LOG(log_type, log_fmt, log_arg...) \
172 do { \
173 uniot_log_printf(PSTR("[" #log_type "][%lu][" __FILE__ ":" \
174 UNIOT_LOG_STRINGIFY(__LINE__) "][%s] " log_fmt "\n"), \
175 millis(), __FUNC_NAME__, ##log_arg); \
176 } while (0)
177
188#define UNIOT_LOG_IF(log_type, log_cond, log_fmt, log_arg...) \
189 do { \
190 if (log_cond) \
191 UNIOT_LOG(log_type, log_fmt, ##log_arg); \
192 } while (0)

234static inline int
235uniot_log_printf(PGM_P format, ...) {
236 va_list arg;
237 va_start(arg, format);
238 char buf[UNIOT_LOG_BUF_SIZE];
239 int len = vsnprintf_P(buf, sizeof(buf), format, arg);
240 va_end(arg);
241 UNIOT_LOG_PRINT(buf);
242 if (sUniotLogSink) sUniotLogSink(buf);
243 if (len >= (int)sizeof(buf))
244 UNIOT_LOG_PRINT(" [...]\n");
245 return len;
246}
