154 void forceSync() {
155 _reconfigure();
156 auto epoch = mSNTP.getNtpTime();
157 if (epoch) {
158 _setTime(epoch);
159 } else {
160 CoreEventEmitter::emitEvent(uniot::events::date::Topic::TIME, uniot::events::date::Msg::SYNC_FAILED);
161 }
162 }

164 private:
171 Date() : CBORStorage("date.cbor") {
172 // A sync is reported from contexts where storing or emitting events is unsafe. On ESP8266
173 // the core runs settimeofday_cb from inside yield(), including the yields LittleFS makes
174 // while it writes, so a store there nests one filesystem operation inside another. On
175 // ESP32 the SNTP callback runs on the network task, concurrently with the loop. The
176 // callbacks therefore only trigger mTaskSync, and the store and the event run on the
177 // scheduler. This also makes a forced sync store the new time: SimpleNTP reports before
178 // forceSync() has applied it.
179 mTaskSync = TaskScheduler::make([](SchedulerTask &, short) {
180 Date::getInstance()._timeSyncCallback();
181 });
182
183#if defined(ESP8266)
184 settimeofday_cb([](bool) {
185 Date::getInstance().mTaskSync->trigger();
186 });
187#elif defined(ESP32)
188 sntp_set_sync_mode(SNTP_SYNC_MODE_IMMED);
189 sntp_set_time_sync_notification_cb([](struct timeval *) {
190 Date::getInstance().mTaskSync->trigger();
191 });
192#endif
193 mSNTP.setSyncTimeCallback([](time_t) {
194 Date::getInstance().mTaskSync->trigger();
195 });
196
197 _reconfigure();
198
199 this->restore();
200 }
201
208 void _timeSyncCallback() {
209 execute(0);
210 UNIOT_LOG_INFO("time synchronized");
211 CoreEventEmitter::emitEvent(uniot::events::date::Topic::TIME, uniot::events::date::Msg::SYNCED);
212 }
213
223 bool _setTime(time_t epoch) {
224#if defined(ESP8266)
225 tune_timeshift64(epoch * 1000000ULL);
226#elif defined(ESP32)
227 timeval tv = {epoch, 0};
228 if (settimeofday(&tv, nullptr) != 0) {
229 UNIOT_LOG_ERROR("failed to set system time");
230 return false;
231 }
232#endif
233 return true;
234 }
235
241 void _reconfigure() {
242 configTime(0, 0, SimpleNTP::servers[0], SimpleNTP::servers[1], SimpleNTP::servers[2]);
243 }
244
250 time_t _now() {
251 return time(nullptr);
252 }
253
262 String _getFormattedTime() {
263 tm tm;
264 auto currentEpoc = this->_now();
265 localtime_r(&currentEpoc, &tm);
266
267 char formattedTime[72] = {0};
268 snprintf(formattedTime, sizeof(formattedTime), "%04d-%02d-%02d %02d:%02d:%02d",
269 tm.tm_year + 1900, tm.tm_mon + 1, tm.tm_mday,
270 tm.tm_hour, tm.tm_min, tm.tm_sec);
271
272 return String(formattedTime);
273 }
