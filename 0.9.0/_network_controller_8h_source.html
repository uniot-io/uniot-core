119 NetworkController(NetworkScheduler &network,
120 uint8_t pinBtn = UINT8_MAX,
121 uint8_t activeLevelBtn = LOW,
122 uint8_t pinLed = UINT8_MAX,
123 uint8_t activeLevelLed = HIGH,
124 uint8_t maxRebootCount = UNIOT_WIFI_REBOOT_RESET_COUNT,
125 uint32_t rebootWindowMs = UNIOT_WIFI_REBOOT_WINDOW_MS)
126 : CBORStorage("ctrl.cbor"),
127 mpNetwork(&network),
128 mNetworkLastState(events::network::Msg::SUCCESS),
129 mClickCounter(0),
130 mPinLed(pinLed),
131 mActiveLevelLed(activeLevelLed),
132 mMaxRebootCount(maxRebootCount),
133 mRebootWindowMs(rebootWindowMs),
134 mRebootCount(0) {
135 if (pinLed != UINT8_MAX) {
136 pinMode(mPinLed, OUTPUT);
137 }
138 if (pinBtn != UINT8_MAX) {
139 mpConfigBtn = MakeUnique<Button>(pinBtn, activeLevelBtn, 30, [&](Button *btn, Button::Event event) {
140 switch (event) {
141 case Button::LONG_PRESS:
142 if (mClickCounter > 3)
143 mpNetwork->forget();
144 else
145 mpNetwork->reconnect();
146 break;
147 case Button::CLICK:
148 if (!mClickCounter) {
149 mpTaskResetClickCounter->attach(5000, 1);
150 }
151 mClickCounter++;
152 default:
153 break;
154 }
155 });
156 }
157 _initTasks();
158 _checkAndHandleReboot();
159 CoreEventListener::listenToEvent(events::network::Topic::CONNECTION);

210 virtual void onEventReceived(unsigned int topic, int msg) override {
211 if (events::network::Topic::CONNECTION == topic) {
212 int lastState = _resetNetworkLastState(msg);
213 switch (msg) {
214 case events::network::Msg::ACCESS_POINT:
215 if (lastState != events::network::Msg::FAILED) {
216 statusWaiting();
217 }
218 break;
219 case events::network::Msg::SUCCESS:
220 statusIdle();
221 break;
222 case events::network::Msg::CONNECTING:
223 statusBusy();
224 break;
225 case events::network::Msg::DISCONNECTED:
226 if (lastState != events::network::Msg::CONNECTING) {
227 // If previous state was CONNECTING, most likely there was a manual reconnect request
228 mpNetwork->reconnect();
229 }
230 break;
231 case events::network::Msg::AVAILABLE:
232 mpNetwork->reconnect();
233 break;
234 case events::network::Msg::FAILED:
235 statusAlarm();
236 mpNetwork->config();
237 break;
238 default:
239 break;
240 }
241 }
242 }

337 void _initTasks() {
338 mpTaskSignalLed = TaskScheduler::make([&](SchedulerTask &self, short t) {
339 static bool signalLevel = true;
340 signalLevel = (!signalLevel && t);
341 CoreEventListener::emitEvent(events::network::Topic::WIFI_STATUS_LED, signalLevel);
342
343 if (_hasLed()) {
344 digitalWrite(mPinLed, signalLevel ? mActiveLevelLed : !mActiveLevelLed);
345 }
346 });
347
348 if (_hasButton()) {
349 mpTaskConfigBtn = TaskScheduler::make(*mpConfigBtn);
350 mpTaskResetClickCounter = TaskScheduler::make([&](SchedulerTask &self, short t) {
351 UNIOT_LOG_DEBUG("click counter: %d", mClickCounter);
352 mClickCounter = 0;
353 });
354 }
355
356 mpTaskResetRebootCounter = TaskScheduler::make([&](SchedulerTask &self, short t) {
357 mRebootCount = 0;
358 NetworkController::store();
359 });
360 }
361
370 void _checkAndHandleReboot() {
371 if (!_hasRebootReset()) {
372 return; // reboot-reset disabled: skip restore, increment, and store entirely
373 }
374 NetworkController::restore();
375 mRebootCount++;
376 if (mRebootCount >= mMaxRebootCount) {
377 mpTaskResetRebootCounter->detach();
378 mpNetwork->forget();
379 mRebootCount = 0;
380 }
381 NetworkController::store();
382 }

393 int _resetNetworkLastState(int newState) {
394 auto oldState = mNetworkLastState;
395 mNetworkLastState = newState;
396 return oldState;
397 }
