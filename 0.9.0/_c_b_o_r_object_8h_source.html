721 Bytes _build(cn_cbor *cb, bool visitSiblings = true) const {
722 auto calculated = cn_cbor_encoder_write(NULL, 0, 0, cb, visitSiblings);
723 Bytes bytes(nullptr, calculated);
724 auto written = bytes.fill([&](uint8_t *buf, size_t size) {
725 auto actual = cn_cbor_encoder_write(buf, 0, size, cb, visitSiblings);
726 if (actual < 0) {
727 UNIOT_LOG_ERROR("%s", "CBORObject build failed, buffer size too small");
728 return 0;
729 }
730 return actual;
731 });
732
733 bytes.prune(written);
734 return bytes;
735 }
